정수 : int, 	문자열 : varchar(숫자),	중복,NULL못하게끔 : PRIMARY KEY

테이블 만들기 : CREATE TABLE ~(
	NUM INT PRIMARY KEY AUTO_INCREMENT
	, TITLE VARCHAR(30)
);

데이터 조회 : SELECT * FROM 테이블명 WHERE 조건 AND(OR) 조건;
데이터 삽입 : INSERT INTO 테이블명 (컬럼명,생략가능) VALUES (저장할데이터);
데이터 삭제 : DELETE FROM 테이블명 WHERE 조건;
데이터 수정 : UPDATE 테이블명 SET 바꿀 컬럼 = '바뀐데이터' WHERE 안바뀌는 데이터;

VARCHAR(숫자) : 최대 숫자 만큼 문자 가능

WHERE ~ IS NULL : 널인거만 조회
WHERE ~ IS NOT NULL : 널이 아닌거만 조회 (0포함)
~ BETWEEN A AND B : 조건이 A와 B사이 
~ IN(100, 200, 300, ...) : 조건이 100 OR 200 OR 300 OR ....인것들

확정 : COMMIT;
취소 : ROLLBACK;

오름차순 : ORDER BY ~ ASC;
내림차순 : ORDER BY ~ DESC;

WHERE ~ LIKE ' %A% ' : ___A___인글자  ('_재_' : 재 앞뒤로 한글자씩 있는것)

YEAR(날짜) >=2021 : 2021년 이후라는 조건

BOARD_NUM INT PRIMARY KEY AUTO_INCREMENT
 : 정수 보드넘은 중복이나 NULL 안되고, 자동으로 1씩 올라간다
READ_CNT INT DEFAULT 0
 : 리드CNT는 0을 기본값으로 가진다 (아무것도 안해도 0이 들어감)
CREATE_DATE DATETIME DEFAULT NOW()
 : 날짜,시간을 현재를 기본값으로 가진다 (아무것도 안해도 현재 날짜,시간 들어감)

DROP TABLE 테이블명; : 테이블을 지우는 명령어

NOW() : 현재시간
SYSDATE() : 시스템시간


중복 제거 (DISTINCT)

CEIL() = 올림
FLOOR() = 내림
ROUND() = 반올림   ROUND(숫자, 소수점자리)
TRUNCATE(숫자, 소수점자리) : 소수점자리 까지 버림
MOD연산자(나머지 연산)

SUBSTR(A,B), SUBSTRING()	A번째 부터 B개 추출
UPPER(문자) : 문자를 대문자로 변경
LOWER(문자) : 문자를 소문자로 변경
TRIM(문자), LTRIM(문자), RTRIM(문자) : 공백제거
CHAR_LENGTH(), LENGTHB() : 글자수 및 크기 확인
CONCAT('A', 'B', ....) : 문자나열
LPAD(문자, 글자수, 채울문자)
RPAD(문자, 글자수, 채울문자)
REPLACE() : 문자 대체

IF(조건, 참일때값, 거짓일때값);
IFNULL(데이터, NULL일때 대체할 데이터)   IFNULL() = NVL()

INNER JOIN (DEPTNO가 공통)
SELECT EMPNO, ENAME, JOB, E.DEPTNO, DNAME
FROM EMP E, DEPT D
WHERE JOB != '사원' 
AND E.DEPTNO = D.DEPTNO; 

INNER JOIN ANSI 문법
SELECT EMPNO, ENAME, JOB, E.DEPTNO, DNAME
FROM EMP E INNER JOIN  DEPT D
ON E.DEPTNO = D.DEPTNO; 
WHERE JOB != '사원'
